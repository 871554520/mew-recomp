# 已知的「函数中间标签」（不是函数入口，是被间接跳转命中的目标）
#
# 大部分标签由 tools/mine_jump_tables.py 从跳转表**静态挖出**（每次构建自动生成
# generated/jump_table_labels.txt），不用维护在这里。这个文件只留静态挖不到的那几个。
#
# 机制回顾：MSVC 把 memcpy/memset 这类长度分派例程编成
#     lea r10,[__ImageBase]
#     cmp r8,0xf ; ja big
#     mov r9d,[r10+rax*4+TABLE] ; add r9,r10 ; jmp r9
# 目标落在**本函数内部**，我们的 dispatcher 只认函数入口 → 命中不了 → 记 miss
# 并当场 return → 被调函数走不到自己的 ret → 返回地址滞留栈上 → 上层收尾寄存器
# 整体错位一格（rbx 被读成返回地址，见 d286ae3）。
#
# 这些地址必须通过 `--seeds-inside` 进入（addExtraEntriesAllowInside）；
# 走 `--seeds` 会被 addExtraEntriesImpl 的 inside_known 判断静默丢弃。
#
# 注意不要往这里塞「共享尾声」（add rsp,0x20; pop rbx; ret 这类）：
# 那种入口会跳过 prologue 导致栈错位，只能由 lifter 的 LIFT_FRAGMENT 在函数内处理。
#
# 来源：discovered_entries.txt（runtime 记录，非跳转表类）
#   0x140d6adc8  caller 0x140d3e654   CFG guard 蹦床之后真正的目标
#   0x140d1ba18  caller 0x1409694c0   0x140d1ba10 的 +8（MSVC 把两个叶子函数
#                                      打包进同一条 .pdata，前 8 字节是 xor eax,eax; ret）
0x140d6adc8
0x140d1ba18

# SEH 恢复点专用的 fragment 入口：runner 的 resume 走 dispatch_indirect，
# 而它只认函数入口，函数中间地址会 FATAL。SKIP_INIT 要把异常后的执行
# 接回 _initterm(0x140d6aec8) 循环里的 0x140d6aef5（call rax 的返回地址，
# 异常帧链 depth=6 实测就是这个值），所以必须让它成为可派发的 fragment。
0x140d6aef5

# _initterm(0x140d6aec8) 的**循环头** 0x140d6aedd。
# 反汇编（0x41 字节，全部解出来）：
#   0x140d6aec8  cmp rcx,rdx / je 0x140d6af08
#   0x140d6aecd  mov [rsp+8],rbx / push rdi / sub rsp,0x20 / mov rdi,rdx / mov rbx,rcx
#   0x140d6aedd  mov rax,[rbx]        <-- 循环头
#   0x140d6aee0  test rax,rax / je 0x140d6aef5  （NULL 项直接跳过）
#   0x140d6aee5  movabs r10,0x85f13e9656da4870 / call qword [CFG槽]
#   0x140d6aef5  add rbx,8 / cmp rbx,rdi / jne 0x140d6aedd    <-- 回跳循环头
#   0x140d6aefe  mov rbx,[rsp+0x30] / add rsp,0x20 / pop rdi / ret
# 上一轮（a64051b）实测：resume 进 0x140d6aef5 这个 fragment 以后，
# `jne 0x140d6aedd` 落到了 fragment 外 → dispatch miss →
#   [FATAL] Unmapped indirect target: 0x140d6aedd ... in_fn=0x140d6aef5 hits=2
# 循环一次都转不起来。把循环头也变成 fragment 就能来回跑。
# 收尾段（add rsp,0x20 / pop rdi / ret）在 0x140d6aedd 这个 fragment 体内，
# 所以从循环头进来的那一趟能把 rsp 正常复原；不要给 0x140d6aefe 另开入口。
0x140d6aedd

# run 45965655068 运行期发现（discovered_entries.txt：`0x140bc5e04 # caller=0x140bc5d50`）。
# 0x140bc5e04 是紧邻 0x140bc5d50 的下一条 .pdata 入口（0x140bc5d50..0x140bc5e04 结束于它），
# 开头 `48 89 5c 24 70` = mov [rsp+0x70],rbx —— **没有自己的序言**，
# 属于同一族「共享序言/尾声」：真函数体跨了这两条 .pdata，只有前一段有 sub rsp。
# 但全镜像里扫不到任何 jcc/jmp/call 直接指向它（跳转表间接命中），
# 所以 merge_shared_frame_size 的「jcc + 紧邻」判定抓不到，只能在这里显式登记。
# 登记成 fragment 后，间接跳转才能命中，而不是 dispatch miss 后当场 return、把返回地址留在栈上。
0x140bc5e04
# run 45991101871 的 FATAL：0x140be07e4 (nearest_lo=0x140be07a0 delta=68)
# 是 0x140be07a0 函数体内部的标签，但被当成函数指针存进了 vtable（外部间接调用），
# 所以必须登记成 fragment；只按 .pdata 入口是找不到它的。
0x140be07e4
# run 46021919377（arm64-verify 的 Stage 4 nested golden）里冒出来的新 FATAL：
#   [FATAL] Unmapped indirect target: 0x140bb067b (nearest_lo=0x0) in_fn=0x140bb0660 hits=1
# 0x140bb067b 是 0x140bb0660 函数体内部 +0x1b 的**指令边界**，逐字节核过：
#   0x140bb0660: 48 89 6c 24 18   mov [rsp+0x18],rbp
#                ...
#               0f 84 96 00 00 00 je 0x140bb0705
#   0x140bb0679: b1 01            mov cl,1
#   0x140bb067b: 48 89 5c 24 30   mov [rsp+0x30],rbx   ← 被当成调用目标的那条
# 即「一条 1 字节指令之后的标签」。nested PoC 是单函数 lift，不会跑 reflow 那一步，
# 所以 g_needed_entries 收不到它 → dispatch miss。显式登记成 fragment 入口即可命中。
0x140bb067b
